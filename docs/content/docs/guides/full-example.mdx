This guide builds a standalone Authsome server on PostgreSQL with email and password sign-in, social login, MFA and organizations.

## 1. Project setup

```bash
mkdir authsome-server && cd authsome-server
go mod init myapp/authsome-server
go get github.com/xraph/authsome@latest github.com/xraph/chronicle@latest \
  github.com/xraph/warden@latest github.com/xraph/grove@latest \
  github.com/xraph/grove/drivers/pgdriver@latest
```

## 2. Docker Compose for dependencies

```yaml
# docker-compose.yml
version: "3.9"
services:
  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_USER: authsome
      POSTGRES_PASSWORD: secret
      POSTGRES_DB: authsome
    ports:
      - "5432:5432"
    volumes:
      - pgdata:/var/lib/postgresql/data

  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"

  mailpit:
    image: axllent/mailpit:latest
    ports:
      - "1025:1025"   # SMTP
      - "8025:8025"   # Web UI

volumes:
  pgdata:
```

Start the dependencies:

```bash
docker compose up -d
```

## 3. Full main.go

```go
package main

import (
    "context"
    "errors"
    "fmt"
    "log"
    "net/http"
    "os"
    "os/signal"
    "syscall"
    "time"

    "github.com/xraph/chronicle"
    chstore "github.com/xraph/chronicle/store"
    chmemory "github.com/xraph/chronicle/store/memory"
    "github.com/xraph/forge"
    "github.com/xraph/grove"
    "github.com/xraph/grove/drivers/pgdriver"
    "github.com/xraph/warden"
    wmemory "github.com/xraph/warden/store/memory"

    "github.com/xraph/authsome"
    "github.com/xraph/authsome/api"
    "github.com/xraph/authsome/bridge"
    "github.com/xraph/authsome/bridge/chronicleadapter"
    "github.com/xraph/authsome/bridge/maileradapter"
    "github.com/xraph/authsome/middleware"
    "github.com/xraph/authsome/plugins/mfa"
    "github.com/xraph/authsome/plugins/organization"
    "github.com/xraph/authsome/plugins/password"
    "github.com/xraph/authsome/plugins/social"
    pgstore "github.com/xraph/authsome/store/postgres"
)

func main() {
    ctx, cancel := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
    defer cancel()

    // Database.
    pg := pgdriver.New()
    if err := pg.Open(ctx, os.Getenv("DATABASE_URL")); err != nil {
        log.Fatal("open db: ", err)
    }
    db, err := grove.Open(pg)
    if err != nil {
        log.Fatal("open db: ", err)
    }
    defer db.Close()

    // Audit trail and authorization (both required). In memory here;
    // give them persistent stores in production.
    trail, err := chronicle.New(chronicle.WithStore(chstore.NewAdapter(chmemory.New())))
    if err != nil {
        log.Fatal(err)
    }
    authz, err := warden.NewEngine(warden.WithStore(wmemory.New()))
    if err != nil {
        log.Fatal(err)
    }

    // Mailer: Resend in production, Mailpit over SMTP in development.
    // The core engine sends no mail itself; the email plugin and your own
    // code send through this bridge (eng.Mailer()).
    var mailer bridge.Mailer
    if apiKey := os.Getenv("RESEND_API_KEY"); apiKey != "" {
        mailer = maileradapter.NewResendMailer(apiKey, os.Getenv("MAIL_FROM"))
    } else {
        mailer = maileradapter.NewSMTPMailer("localhost", "1025", "", "", "noreply@example.com")
    }

    cfg := authsome.DefaultConfig()
    cfg.Session.TokenTTL = time.Hour
    cfg.Session.RefreshTokenTTL = 30 * 24 * time.Hour
    cfg.Password.MinLength = 12
    cfg.RateLimit.SignInLimit = 5
    cfg.RateLimit.SignUpLimit = 3
    cfg.Lockout.MaxAttempts = 5
    cfg.Lockout.LockoutDurationSeconds = 900

    engine, err := authsome.NewEngine(
        authsome.WithConfig(cfg),
        authsome.WithStore(pgstore.New(db)),
        authsome.WithDriverName("pg"), // lets plugins pick their PostgreSQL migrations
        authsome.WithDB(db),           // lets plugins such as mfa and social store their data in PostgreSQL
        authsome.WithChronicle(chronicleadapter.New(trail)),
        authsome.WithWarden(authz),
        authsome.WithMailer(mailer),
        authsome.WithPlugin(password.New()),
        authsome.WithPlugin(mfa.New(mfa.Config{Issuer: "Acme"})),
        authsome.WithPlugin(organization.New()),
        authsome.WithPlugin(social.New(social.Config{
            Domain: "http://localhost:8080",
            Providers: []social.Provider{
                social.NewGoogleProvider(social.ProviderConfig{
                    ClientID:     os.Getenv("GOOGLE_CLIENT_ID"),
                    ClientSecret: os.Getenv("GOOGLE_CLIENT_SECRET"),
                }),
            },
        })),
        authsome.WithBootstrap(authsome.WithBootstrapAppName("Acme")),
    )
    if err != nil {
        log.Fatal("create engine: ", err)
    }

    // Start runs the migrations for the core and every plugin.
    if err := engine.Start(ctx); err != nil {
        log.Fatal("start engine: ", err)
    }
    defer engine.Stop(context.Background())

    router := forge.NewRouter()
    router.Use(middleware.AuthMiddleware(engine.ResolveSessionByToken, engine.ResolveUser, nil))
    if err := api.New(engine).RegisterRoutes(router); err != nil {
        log.Fatal(err)
    }
    // Plugins mount their own routes (/v1/orgs, /v1/mfa, /v1/social and so on).
    for _, rp := range engine.Plugins().RouteProviders() {
        if err := rp.RegisterRoutes(router); err != nil {
            log.Fatal(err)
        }
    }

    mux := http.NewServeMux()
    mux.Handle("/", router.Handler())
    mux.HandleFunc("GET /health", func(w http.ResponseWriter, r *http.Request) {
        if err := engine.Health(r.Context()); err != nil {
            http.Error(w, "unhealthy", http.StatusServiceUnavailable)
            return
        }
        fmt.Fprint(w, "ok")
    })

    addr := ":" + envOr("PORT", "8080")
    srv := &http.Server{Addr: addr, Handler: mux, ReadHeaderTimeout: 10 * time.Second}

    go func() {
        log.Printf("authsome listening on %s, app_id=%s", addr, engine.PlatformAppID())
        if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
            log.Fatal("serve: ", err)
        }
    }()

    <-ctx.Done()
    shutdownCtx, shutdownCancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer shutdownCancel()
    _ = srv.Shutdown(shutdownCtx)
}

func envOr(key, fallback string) string {
    if v := os.Getenv(key); v != "" {
        return v
    }
    return fallback
}
```

A few things to notice:

- Chronicle and Warden are required. The engine refuses to start without them. They run in memory here, so the audit trail and role grants reset on restart; give them persistent stores before production.
- `engine.Start` runs the migrations for the core and every registered plugin.
- `WithBootstrap` creates the platform app on first start. The log line prints its ID, which clients send as `app_id` (or as an `X-Publishable-Key` header).
- Rate limits and lockout are on by default. With no limiter or tracker passed, the engine keeps their counters in the store, so every instance sharing the database shares the limits.

## 6. Testing the server

Use the `app_id` from the startup log line.

### Sign up

```bash
curl -X POST http://localhost:8080/v1/signup \
  -H "Content-Type: application/json" \
  -d '{
    "app_id": "aapp_...",
    "email": "jane@example.com",
    "password": "SecurePass1!",
    "first_name": "Jane",
    "last_name": "Doe"
  }'
```

The response has `session_token`, `refresh_token`, `expires_at` and `user`.

### Verify the email, then sign in

Sign-in refuses an unverified address while `auth.require_email_verification` is on (the default). Sign-up issues a 6-digit verification code, but Authsome doesn't email it on its own: the `notification` plugin delivers it through a Herald bridge, and the mailer above isn't involved. Until you set that up, you have two options:

- Deliver the code yourself. `engine.SendEmailVerification(ctx, user)` issues a fresh code and returns it; the client then posts `{"email": "jane@example.com", "code": "123456"}` to `POST /v1/verify-email`.
- Turn the check off while you develop:

  ```go
  _ = engine.Settings().Set(ctx, "auth.require_email_verification",
      json.RawMessage(`false`), settings.ScopeGlobal, "", "", "", "dev")
  ```

Then sign in:

```bash
curl -X POST http://localhost:8080/v1/signin \
  -H "Content-Type: application/json" \
  -d '{
    "app_id": "aapp_...",
    "email": "jane@example.com",
    "password": "SecurePass1!"
  }'
```

### Get current user

```bash
curl http://localhost:8080/v1/me \
  -H "Authorization: Bearer <session_token>"
```

### Refresh token

```bash
curl -X POST http://localhost:8080/v1/refresh \
  -H "Content-Type: application/json" \
  -d '{"refresh_token": "<refresh_token>"}'
```

### Sign out

```bash
curl -X POST http://localhost:8080/v1/signout \
  -H "Authorization: Bearer <session_token>"
```

### List organizations

The `organization` plugin adds its routes under `/v1/orgs`:

```bash
curl http://localhost:8080/v1/orgs \
  -H "Authorization: Bearer <session_token>"
```

## 7. Adding plugins

Each plugin registers its own routes and migrations. Add more with `authsome.WithPlugin`:

```go
import (
    "github.com/xraph/authsome/plugins/magiclink"
    "github.com/xraph/authsome/plugins/passkey"
    "github.com/xraph/authsome/plugins/social"
)

authsome.WithPlugin(social.New(social.Config{
    Domain: "https://auth.example.com",
    Providers: []social.Provider{
        social.NewGoogleProvider(social.ProviderConfig{
            ClientID:     os.Getenv("GOOGLE_CLIENT_ID"),
            ClientSecret: os.Getenv("GOOGLE_CLIENT_SECRET"),
        }),
        social.NewGitHubProvider(social.ProviderConfig{
            ClientID:     os.Getenv("GITHUB_CLIENT_ID"),
            ClientSecret: os.Getenv("GITHUB_CLIENT_SECRET"),
        }),
    },
})),
authsome.WithPlugin(passkey.New(passkey.Config{
    RPDisplayName: "My App",
    RPID:          "example.com",
    RPOrigins:     []string{"https://example.com"},
})),
authsome.WithPlugin(magiclink.New()),
```

## 8. Production considerations

- Use a connection pooler (PgBouncer) for PostgreSQL in production
- Give Chronicle and Warden persistent stores
- Set `RESEND_API_KEY` and `MAIL_FROM` for transactional email
- Add TLS termination via a reverse proxy (nginx, Caddy, or a cloud load balancer)
- For multi-service deployments, use the [Forge Extension](/docs/guides/forge-extension) instead
